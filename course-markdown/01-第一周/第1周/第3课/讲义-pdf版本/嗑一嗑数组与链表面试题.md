    public void moveZeroes(int[] nums) {
        if (nums == null || nums.length < 2) {
            return;
        }
        //从前到后非零元素的指针
        int p1 = 0;
        for (int i =0; i < nums.length; i++) {
            if (nums[i] != 0) {
                nums[p1] = nums[i];
                p1++;
            }
        }
        while (p1 < nums.length) {
            nums[p1] = 0;
            p1++;
        }
    }
}

            return false;

        ListNode slow = head;

                break;

private ListNode reverse(ListNode head){
    ListNode pre = null;
    ListNode curr = head;
    while (curr!=null) {
